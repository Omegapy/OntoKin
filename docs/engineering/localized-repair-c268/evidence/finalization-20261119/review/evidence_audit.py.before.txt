#!/usr/bin/env python3
"""Verify C268 candidate cohort, payload, retained failures and review identity."""
import argparse
from collections import Counter
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess
import sys


def ident(raw):
    return {"bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest()}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--repo", type=Path, required=True)
    parser.add_argument("--identity-repo", type=Path, required=True)
    parser.add_argument("--cohort", type=Path, required=True)
    parser.add_argument("--copied-cohort", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    local = "52af987fd5705c034bf264e2fe3fbc9fa1f146ae"
    published = "e6f55b73fa71953aa648bcd34711a6796a7346b6"
    base = "409ef18e0cb992ad1ab771c1258e989d5ef1c765"
    tree = "d94436a22552db903b8016f9d035985a7d8b6752"
    def git(*arguments, identity_repo=False):
        return subprocess.check_output(["git", *arguments], cwd=args.identity_repo if identity_repo else args.repo)
    report = {"schema": "ontokin.c268.final-review-evidence-audit/1", "reviewer": "/root/review_c268_final",
              "at_utc": datetime.now(timezone.utc).isoformat(),
              "command": [sys.executable, "-B", str(Path(__file__).resolve()), *sys.argv[1:]],
              "script": ident(Path(__file__).read_bytes()), "local_candidate": local,
              "published_candidate": published, "tree": tree, "base": base, "assertions": {}}
    assertions = report["assertions"]
    assertions["local_tree"] = git("rev-parse", local + "^{tree}").decode().strip() == tree
    assertions["published_tree"] = git("rev-parse", published + "^{tree}", identity_repo=True).decode().strip() == tree
    assertions["whole_tree_equality"] = git("diff", "--name-only", local, published, identity_repo=True) == b""
    assertions["base_is_ancestor"] = subprocess.run(["git", "merge-base", "--is-ancestor", base, published], cwd=args.identity_repo).returncode == 0
    report["changed_paths"] = git("diff", "--name-only", base, local).decode().splitlines()
    assertions["cumulative_scope_only_C268_and_task"] = all(path.startswith(("docs/engineering/localized-repair-c268/", "docs/workflow/tasks/C268/")) for path in report["changed_paths"])
    assertions["review_clone_clean"] = git("status", "--porcelain=v1", "--untracked-files=no") == b""
    cohort_path = args.cohort / "CHECKS.json"
    cohort = json.loads(cohort_path.read_text())
    report["cohort_receipt"] = {"path": str(cohort_path), **ident(cohort_path.read_bytes())}
    report["copied_cohort_receipt"] = {"path": str(args.copied_cohort / "CHECKS.json"), **ident((args.copied_cohort / "CHECKS.json").read_bytes())}
    assertions["cohort_copy_matches"] = cohort_path.read_bytes() == (args.copied_cohort / "CHECKS.json").read_bytes()
    for name in ("source_before", "source_after"):
        source = cohort[name]
        assertions[name + "_matches"] = (source["commit"], source["tree"], source["tracked_dirty_status"], source["all_dirty_status"]) == (published, tree, "", "")
    assertions["cohort_complete"] = cohort["complete"] is True and cohort["passed"] is True and cohort["completed_check_count"] == 12 and cohort["total_methods"] == 282
    assertions["cohort_inventory_stable"] = cohort["input_files_before"] == cohort["input_files_after"] and cohort["input_identity_unchanged"] is True
    report["input_file_count"] = len(cohort["input_files_before"])
    mismatches = []
    for entry in cohort["input_files_before"]:
        raw = git("show", local + ":" + entry["path"])
        if ident(raw) != {key: entry[key] for key in ("bytes", "sha256")}:
            mismatches.append(entry["path"])
    report["candidate_input_mismatches"] = mismatches
    assertions["all_cohort_inputs_equal_reviewed_git_blobs"] = not mismatches
    report["output_file_count"] = len(cohort["output_files"])
    output_checks = []
    for entry in cohort["output_files"]:
        expected = {key: entry[key] for key in ("bytes", "sha256")}
        actual = ident((args.cohort / entry["path"]).read_bytes())
        copied = ident((args.copied_cohort / entry["path"]).read_bytes())
        output_checks.append({"path": entry["path"], **actual, "receipt_matches": actual == expected,
                              "copied_payload_matches": copied == expected})
    report["outputs"] = output_checks
    assertions["all_output_receipts_and_copies_match"] = all(row["receipt_matches"] and row["copied_payload_matches"] for row in output_checks)
    runner_path = "docs/engineering/localized-repair-c268/evidence/run_checks.py"
    assertions["runner_equals_reviewed_blob"] = ident(git("show", local + ":" + runner_path)) == {key: cohort["runner"][key] for key in ("bytes", "sha256")}
    assertions["runtime_binary_identity"] = ident(Path(cohort["environment"]["resolved_executable"]["path"]).read_bytes()) == {key: cohort["environment"]["resolved_executable"][key] for key in ("bytes", "sha256")}
    report["environment"] = cohort["environment"]
    report["groups"] = [{key: row[key] for key in ("name", "command", "methods", "result", "exit_code", "failure")} for row in cohort["checks"]]
    assertions["all_groups_pass"] = all(row["result"] == "passed" and row["exit_code"] == 0 and row["failure"] is None and all(row["assertions"].values()) for row in cohort["checks"])
    controls = json.loads((args.cohort / "c268-controls/results.json").read_text())
    counts = Counter(row["outcome"] for row in controls["cases"])
    api_counts = Counter(row["api"] for row in controls["cases"])
    report["C268_actual_outcomes"] = dict(counts)
    report["C268_actual_apis"] = dict(api_counts)
    assertions["C268_full_ledger_accounting"] = (len(controls["cases"]) == controls["apiCalls"] == 131
        and api_counts["historical-reader"] == controls["historicalCalls"] == 121
        and sum(value for key, value in api_counts.items() if key != "historical-reader") == controls["rawCalls"] == 10
        and counts == Counter(admitted=12, raw_success=9, typed_refusal=110)
        and controls["admittedCalls"] == 12 and controls["rawSuccessfulCalls"] == 9 and controls["typedRefusals"] == 110
        and controls["untypedExceptions"] == 0)
    assertions["C268_zero_skips_errors_failures"] = controls["testMethods"] == 35 and controls["passed"] is True and controls["failures"] == controls["errors"] == controls["skipped"] == []
    control_mismatches = []
    for path, expected in controls["implementationAndControlInputs"].items():
        if ident(git("show", local + ":" + path)) != expected:
            control_mismatches.append(path)
    report["control_input_mismatches"] = control_mismatches
    assertions["all_control_input_identities_equal_candidate"] = not control_mismatches
    report["permission_observations"] = controls["notes"]
    prior = "734789521f82248d39b623c33aae9d3284ab4bce"
    prefix = "docs/engineering/localized-repair-c268/"
    preserved = ["evidence/control-author/run-01", "evidence/control-author/run-02", "evidence/control-author/run-03",
                 "evidence/control-author/discovery-01", "evidence/candidate"]
    assertions["historical_failed_and_predecessor_checks_unchanged"] = git("diff", "--name-only", prior, local, "--", *[prefix + value for value in preserved]) == b""
    failed = json.loads(git("show", local + ":" + prefix + "evidence/control-author/run-01/results.json"))
    report["preserved_failed_run"] = {"testMethods": failed["testMethods"], "errors": len(failed["errors"]), "failures": len(failed["failures"]), "passed": failed["passed"], "error_kinds": sorted(set(row["traceback"].strip().splitlines()[-1] for row in failed["errors"]))}
    report["corrected_control_delta_numstat"] = git("diff", "--numstat", prior, local, "--", prefix + "test_endpoint_reader.py").decode().strip()
    assertions["prior_reader_snapshot_exact"] = git("show", prior + ":" + prefix + "endpoint_reader.py") == git("show", local + ":" + prefix + "evidence/implementation/endpoint-length-fix/endpoint_reader.py.before.txt")
    report["passed"] = all(assertions.values())
    args.output.write_text(json.dumps(report, indent=2) + "\n")
    print(json.dumps({"passed": report["passed"], "assertions": assertions, "inputs": report["input_file_count"], "outputs": report["output_file_count"], "cohort_receipt": report["cohort_receipt"], "preserved_failed_run": report["preserved_failed_run"]}, indent=2))
    return 0 if report["passed"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
