def digest(value):
    return hashlib.sha256(value).hexdigest()


def source_bundle(endpoint="before"):
    return {"main.ok": (PROOF / (endpoint + ".main.ok")).read_bytes(),
            "hinges.ok": (PROOF / "hinges.ok").read_bytes()}


def repin(bundle):
    result = dict(bundle)
    result["main.ok"], count = re.subn(
        rb'digest "sha256:[a-f0-9]{64}"',
        b'digest "sha256:' + digest(result["hinges.ok"]).encode() + b'"',
        result["main.ok"], count=1)
    assert count == 1, "Independent fixture must retain one import digest"
    return result


def library_edit(bundle, old, new, count=1):
    result = dict(bundle)
    assert old in result["hinges.ok"], old
    result["hinges.ok"] = result["hinges.ok"].replace(old, new, count)
    return repin(result)


def main_edit(bundle, old, new, count=1):
    result = dict(bundle)
    assert old in result["main.ok"], old
    result["main.ok"] = result["main.ok"].replace(old, new, count)
    return result


def block_span(raw, marker):
    """Locate a known fixture declaration, respecting quoted braces."""
    beginning = raw.index(marker)
    cursor = raw.index(b"{", beginning)
    depth, quoted, escaped = 0, False, False
    for pos in range(cursor, len(raw)):
        char = raw[pos]
        if quoted:
            if escaped:
                escaped = False
            elif char == 92:
                escaped = True
            elif char == 34:
                quoted = False
        elif char == 34:
            quoted = True
        elif char == 123:
            depth += 1
        elif char == 125:
            depth -= 1
            if not depth:
                return beginning, pos + 1
    raise AssertionError("Fixture block is unbalanced")


def replace_block(bundle, filename, marker, replacement):
    result = dict(bundle)
    start, end = block_span(result[filename], marker)
    result[filename] = result[filename][:start] + replacement + result[filename][end:]
    return repin(result) if filename == "hinges.ok" else result


def field_change(bundle, field, expression):
    originals = {
        "parent": b"self.pivot", "child": b"self.blade.base",
        "axis": b"vec3(0, 0, 1)", "limits": b"range(-45[deg], 45[deg])",
        "initial": b"0[rad]",
    }
    return library_edit(bundle, field.encode() + b" = " + originals[field] + b";",
                        field.encode() + b" = " + expression + b";")


def replace_helper_keys(bundle, keys):
    entries = [json.dumps(key, ensure_ascii=False).encode() + b": vec3(0[m], 0[m], 0[m])" for key in keys]
    replacement = b"keyed<Vec3<Length>> {" + b",\n".join(entries) + b"}"
    result = replace_block(bundle, "hinges.ok", b"keyed<Vec3<Length>>", replacement)
    if keys and "middle" not in keys:
        result = main_edit(result, b'fingers["middle"]', b"fingers[" + json.dumps(keys[0], ensure_ascii=False).encode() + b"]", count=-1)
    return result


def baseline_ids(first_keys=("lower", "middle", "upper"), second_keys=("bottom", "lower", "middle", "upper", "top")):
    return {("keyed-rack", "demo", root, "fingers", key, "flex")
            for root, keys in (("first", first_keys), ("second", second_keys)) for key in keys}


def canonical_decimal(token, unit):
    """Independent high precision decimal reference, never an observer helper."""
    with decimal.localcontext() as context:
        context.prec = 1000
        value = decimal.Decimal(token)
        if unit == "deg":
            value *= decimal.Decimal.from_float(float.fromhex("0x1.921fb54442d18p+1"))
            value /= decimal.Decimal(180)
        elif unit == "cm":
            value /= decimal.Decimal(100)
        elif unit == "mm":
            value /= decimal.Decimal(1000)
        elif unit not in ("rad", "m"):
            raise AssertionError(unit)
        return float(value)


def field_projection(observation):
    return {tuple(row["id"]): (row["axis"], row["limits"], row["initial"],
                               row["parentSiteId"], row["childSiteId"])
            for row in observation["records"]}


class SourceHingeControls(unittest.TestCase):
    def observe(self, bundle):
        identities = {str(name): {"sha256": digest(raw), "bytes": len(raw)}
                      for name, raw in bundle.items() if isinstance(raw, bytes)} if isinstance(bundle, dict) else {}
        entry = {"test": self.id(), "inputs": identities}
        OBSERVATIONS.append(entry)
        try:
            result = reader.observe_source_hinges(bundle)
        except reader.Fault as fault:
            entry.update(outcome="refused", code=fault.code, detail=str(fault))
            raise
        except Exception as error:
            entry.update(outcome="HOST_EXCEPTION", exception=type(error).__name__, detail=str(error))
            raise
        entry.update(outcome="observed", hingeCount=result.get("hingeCount"), siteCount=result.get("siteCount"))
        return result

    def refused(self, bundle, codes=None):
        with self.assertRaises(reader.Fault) as context:
            self.observe(bundle)
        fault = context.exception
        self.assertIn(fault.code, set(codes) if codes else FAULT_CODES)
        self.assertFalse(getattr(fault, "records", None), "Refusal must not carry a qualified hinge set")
        self.assertIsInstance(fault.sourceHashes, dict)
        self.assertIsInstance(fault.sourceBytes, dict)
        if isinstance(bundle, dict):
            for name, raw in bundle.items():
                if isinstance(name, str) and isinstance(raw, bytes) and name in ("main.ok", "hinges.ok"):
                    self.assertEqual(len(raw), fault.sourceBytes.get(name))
                    if len(raw) <= 65536:
                        self.assertEqual(digest(raw), fault.sourceHashes.get(name))
                    elif name in fault.sourceHashes:
                        self.assertEqual(digest(raw), fault.sourceHashes[name])
        return fault

    def assert_values(self, result, axis=(0.0, 0.0, 1.0), limits=None, initial=0.0):
        limits = (-float.fromhex("0x1.921fb54442d18p-1"), float.fromhex("0x1.921fb54442d18p-1")) if limits is None else limits
        for row in result["records"]:
            self.assertEqual(list(axis), row["axis"])
            self.assertEqual(list(limits), row["limits"])
            self.assertEqual(initial, row["initial"])
            self.assertEqual("hinge", row["kind"])

    def assert_unqualified(self, result):
        for key in TRUTH_FLAGS:
            self.assertIs(False, result[key], key)
        for key in ("acceptedRevision", "nativeVerified", "trialQualified", "worldAlignmentVerified"):
            self.assertFalse(result.get(key), key)

    def witness_bytes(self, witness, bundle):
        self.assertIn(witness["file"], bundle)
        raw = bundle[witness["file"]]
        self.assertEqual(digest(raw), witness["sha256"])
        self.assertIs(type(witness["startByte"]), int)
        self.assertIs(type(witness["endByteExclusive"]), int)
        self.assertLessEqual(0, witness["startByte"])
        self.assertLess(witness["startByte"], witness["endByteExclusive"])
        self.assertLessEqual(witness["endByteExclusive"], len(raw))
        return raw[witness["startByte"]:witness["endByteExclusive"]]

    def test_h01_both_original_source_endpoints(self):
        for endpoint in ("before", "after"):
            with self.subTest(endpoint=endpoint):
                bundle = source_bundle(endpoint)
                result = self.observe(bundle)
                self.assertEqual("ontokin.c265.source-hinge-observation/0.1", result["schema"])
                self.assertEqual("source-hinges", result["kind"])
                self.assertEqual(8, result["hingeCount"])
                self.assertEqual(16, result["siteCount"])
                self.assertEqual(18, len(result["objects"]))
                self.assertEqual(baseline_ids(), {tuple(row["id"]) for row in result["records"]})
                self.assertEqual({("package", "example.hinges", "hinges", "definitions", "finger", "flex")},
                                 {tuple(row["definitionId"]) for row in result["records"]})
                self.assertEqual({name: digest(raw) for name, raw in bundle.items()}, result["sourceHashes"])
                self.assert_values(result)
                self.assert_unqualified(result)

    def test_h01_complete_associations_are_independently_derived(self):
        result = self.observe(source_bundle())
        for row in result["records"]:
            owner = row["id"][:-1]
            root, key = owner[2], owner[4]
            path = root + '.fingers[' + json.dumps(key) + ']'
            self.assertEqual(owner, row["ownerId"])
            self.assertEqual(path + ".flex", row["path"])
            self.assertEqual(owner + ["pivot"], row["parentSiteId"])
            self.assertEqual(owner + ["blade", "base"], row["childSiteId"])
            self.assertEqual(path + ".pivot", row["parentSite"]["path"])
            self.assertEqual(path + ".blade.base", row["childSite"]["path"])
            self.assertEqual(owner, row["parentSite"]["objectId"])
            self.assertEqual(owner + ["blade"], row["childSite"]["objectId"])
            self.assertEqual(["package", "example.hinges", "hinges", "definitions", "finger", "pivot"], row["parentSite"]["definitionId"])
            self.assertEqual(["package", "example.hinges", "hinges", "definitions", "blade", "base"], row["childSite"]["definitionId"])
            for endpoint in (row["parentSite"], row["childSite"]):
                self.assertEqual("hinged.part", endpoint["interface"])
                self.assertEqual(0.0, endpoint["sizeM"])
                self.assertIs(False, endpoint["poseEvaluated"])

    def test_h01_utf8_byte_witnesses_are_exact(self):
        bundle = library_edit(source_bundle(), b"module Hinges", "// π and 🦉 precede declarations\nmodule Hinges".encode())
        result = self.observe(bundle)
        field_expressions = {"parent": b"self.pivot", "child": b"self.blade.base", "axis": b"vec3(0, 0, 1)", "limits": b"range(-45[deg], 45[deg])", "initial": b"0[rad]"}
        for row in result["records"]:
            origin = row["sourceOrigins"]
            self.assertIn(b'joint flex @ "flex": hinge', self.witness_bytes(origin["declaration"], bundle))
            self.assertEqual(b"flex", self.witness_bytes(origin["name"], bundle))
            self.assertIn(b"flex", self.witness_bytes(origin["key"], bundle))
            for name, expected in field_expressions.items():
                self.assertEqual(expected, self.witness_bytes(origin["fields"][name]["expression"], bundle))
                self.assertIn(name.encode() + b" = " + expected, self.witness_bytes(origin["fields"][name]["assignment"], bundle))
            for site in (row["parentSite"], row["childSite"]):
                self.assertIn(b"site ", self.witness_bytes(site["sourceOrigins"]["declaration"], bundle))
                for name in ("pose", "interface", "size"):
                    self.witness_bytes(site["sourceOrigins"]["fields"][name]["expression"], bundle)

    def test_h06_inclusive_boundaries_and_zero_width(self):
        cases = ((b"range(-1[rad], 1[rad])", b"-1[rad]", (-1.0, 1.0), -1.0),
                 (b"range(-1[rad], 1[rad])", b"1[rad]", (-1.0, 1.0), 1.0),

    def test_h07_pi64_single_round_literals_and_ties(self):
        cases = (("0.1", "deg"), ("-45", "deg"), ("180", "deg"), ("0.7853981633974483", "rad"),

    def test_h17_duplicate_names_keys_and_reserved_members_refuse(self):
        start, end = block_span(source_bundle()["hinges.ok"], JOINT)
        declaration = source_bundle()["hinges.ok"][start:end]
        cases = [library_edit(source_bundle(), declaration, declaration + b"\n" + declaration),
                 library_edit(source_bundle(), b'joint flex @ "flex"', b'joint pivot @ "flex"'),

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--evidence-dir", type=Path, help="Fresh output directory; existing paths refuse to prevent overwriting runs")
    arguments = parser.parse_args()
    suite = unittest.defaultTestLoader.loadTestsFromTestCase(SourceHingeControls)
    log = io.StringIO()
    result = unittest.TextTestRunner(stream=log, verbosity=2).run(suite)
    output = log.getvalue()
    sys.stdout.write(output)
    if arguments.evidence_dir:
        arguments.evidence_dir.mkdir(parents=True, exist_ok=False)
        (arguments.evidence_dir / "unittest.log").write_text(output)
        for path in (READER_PATH, Path(__file__)):
            (arguments.evidence_dir / (path.name + ".txt")).write_bytes(INPUT_BYTES[path])
        summary = {
            "schema": "ontokin.c265.independent-source-controls/0.1",
            "createdUTC": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "command": [sys.executable, *sys.argv],
            "python": sys.version, "platform": platform.platform(), "machine": platform.machine(),
            "inputs": {str(path.relative_to(ROOT)): {"sha256": digest(raw), "bytes": len(raw)}
                       for path, raw in INPUT_BYTES.items()},
            "testsRun": result.testsRun, "failures": len(result.failures), "errors": len(result.errors),
            "skipped": len(result.skipped), "passed": result.wasSuccessful(),
            "observations": OBSERVATIONS,
            "coverageBoundary": "Source portions only; artifact comparison, historical admission, full source validity, endpoint poses, native execution and model trials are not run or qualified.",
        }
        (arguments.evidence_dir / "results.json").write_text(json.dumps(summary, indent=2) + "\n")
    return 0 if result.wasSuccessful() else 1


if __name__ == "__main__":
    sys.exit(main())
